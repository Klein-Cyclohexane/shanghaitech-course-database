\titledquestion{Compare and Prove}

For each pair of functions $f(n)$ and $g(n)$, give your answers whether $f(n) = o(g(n))$, $f(n) = \omega(g(n))$ or $f(n) = \Theta(g(n))$.  Give a \textbf{proof} of your answer. 

\textbf{Note 1}: Give your answer in the most precise form. 

\textbf{Note 2}: You'd better try to prove them by calculating limits.

\begin{parts}
    \part[3] $f(n) = {1.01}^n$ and $g(n) = n^{10}$
    \begin{solution}
        $f(n) = \omega(g(n))$ \\
        Proof: \\
        According to the definition of $\omega$, we need to prove that $\lim_{n\to\infty} \frac{{1.01}^n}{n^{10}} = \infty$. \\
        And with the help of L'Hopital's Rule, we can get: \\
        $\lim_{n\to\infty} \frac{{1.01}^n}{n^{10}} = \lim_{n\to\infty} \frac{{1.01}^n \ln(1.01)}{10n^9} \\
        = \lim_{n\to\infty} \frac{{1.01}^n \ln^2(1.01)}{90n^8} = \cdots = \lim_{n\to\infty} \frac{{1.01}^n \ln^{10}(1.01)}{10!} = \infty$ \\
        So, $f(n) = \omega(g(n))$.
    \end{solution}

    \newpage
    \part[4] $f(n) = \log(n!)$ and $g(n) = \log(n^n)$
    \begin{solution}
        $f(n) = \Theta(g(n))$ \\
        Proof: \\
        According to the definition of $\Theta$, we need to prove that $\lim_{n\to\infty} \frac{\log(n!)}{\log(n^n)} = c, 0 < c < \infty$. \\
        And with the help of Stirling's Formula, we can get: \\
        $\lim_{n\to\infty} \frac{\log(n!)}{\log(n^n)} = \lim_{n\to\infty} \frac{\log(\sqrt{2\pi n}(\frac{n}{e})^n)}{\log(n^n)} \\
        = \lim_{n\to\infty} \frac{\log(\sqrt{2\pi n}) + \log((\frac{n}{e})^n)}{n\log n} = \lim_{n\to\infty} \frac{\log(\sqrt{2\pi n}) + n\log(\frac{n}{e})}{n\log n} \\
        = \lim_{n\to\infty} \frac{\log(\sqrt{2\pi n})}{n\log n} + \lim_{n\to\infty} \frac{n\log(\frac{n}{e})}{n\log n} = 0 + 1 = 1$ \\
        So, $f(n) = \Theta(g(n))$.
    \end{solution}
    
    \newpage
    \part[4] $f(n) = n^{1+\varepsilon}, \varepsilon\in \mathbb{R}^+$ and $g(n) = n(\log n)^k, k \in \mathbb{Z}^+$
    \begin{solution}
        $f(n) = \omega(g(n))$ \\
        Proof: \\
        According to the definition of $\omega$, we need to prove that $\lim_{n\to\infty} \frac{n^{1+\varepsilon}}{n(\log n)^k} = \infty$. \\
        $\lim_{n\to\infty} \frac{n^{1+\varepsilon}}{n(\log n)^k} = \lim_{n\to\infty} \frac{n^{\varepsilon}}{(\log n)^k} \\$
        Next, we can use L'Hopital's Rule to get: \\
        $\lim_{n\to\infty} \frac{n^{\varepsilon}}{(\log n)^k} = \lim_{n\to\infty} \frac{\varepsilon n^{\varepsilon}}{k(\log n)^{k-1}} = \cdots = \lim_{n\to\infty} \frac{\varepsilon^k n^\varepsilon}{k!} = \infty$ \\
        So, $f(n) = \omega(g(n))$.
    \end{solution}
\end{parts}
